        \begin{itemize}
          \item The rapid growth of online testing faces severe challenges in maintaining academic integrity and preventing sophisticated cheating.
          \item Traditional proctoring relies heavily on human supervision or cloud solutions, causing high latency and privacy risks.
          \item \textbf{Motivation:} Develop an automated Edge AI (local processing) proctoring system to eliminate network latency, guarantee data privacy, and enable multimodal analysis.
          \item \textbf{Significance:} Provides a robust, secure, and automated monitoring solution for modern educational environments.

%  SLIDE 5 – Aim and Objectives
% ------------------------------------------------------------
\begin{frame}{Objectives}
\begin{itemize}
    \item \textbf{Build 4 Core AI Proctoring Modules:}
    \begin{itemize}
        \item \textbf{YOLOv8:} Detect prohibited items on desks (smartphones, cheat sheets, earphones).
        \item \textbf{MediaPipe Pose + Face Mesh:} Extract 3D landmarks and estimate Pitch/Yaw/Roll head orientation via PnP.
        \item \textbf{FaceNet:} Compute Cosine Similarity between real-time frames and ID images to detect substitute test-takers.
        \item \textbf{VAD + Whisper AI + PhoBERT:} Filter ambient noise and analyze audio keywords for illicit verbal communication.
    \end{itemize}
    \item \textbf{Optimize System Architecture:} Implement Multithreading, Thread-Safe Queues, and Frame Skipping (5 FPS) for smooth local execution.
    \item \textbf{Visualization \& Logging:} Display real-time GUI overlays and record detailed logs in JSON/SQLite.
\end{itemize}
\end{frame}

\begin{frame}{Methodology}
  \begin{methodcols}
    \methodcol{\ding{202}~Problem Setup}{%
      \begin{itemize}
        \item \textbf{Input:} Webcam feed (640x640 numpy matrix) + Audio stream (PyAudio chunks).
        \item \textbf{Output:} Visual violation alert frames, JSON log streams, and local image evidence.
      \end{itemize}%
    }
    \methodcol{\ding{203}~Core Algorithms}{%
      \begin{itemize}
        \item \textbf{PnP / Gaze:} Euler angles (Pitch, Yaw, Roll) via 3D-2D facial landmark mapping.
        \item \textbf{FaceNet:} Cosine similarity threshold ($\ge 0.75$) for identity verification.
        \item \textbf{Whisper + VAD:} Speech-to-text keyword analysis.
      \end{itemize}%
    }
    \methodcol{\ding{204}~System Architecture}{%
      \begin{itemize}
        \item \textbf{Approach:} Multithreaded Edge AI with Frame Skipping (5 FPS).
        \item \textbf{Tools:} Python, OpenCV, MediaPipe, YOLOv8, PyTorch, SQLite.
      \end{itemize}%
    }
  \end{methodcols}
